package com.mohhwebsites.core.models;

import org.apache.sling.api.SlingHttpServletRequest;
import org.apache.sling.api.resource.Resource;
import org.apache.sling.models.annotations.Default;
import org.apache.sling.models.annotations.DefaultInjectionStrategy;
import org.apache.sling.models.annotations.Model;
import org.apache.sling.models.annotations.injectorspecific.ChildResource;
import org.apache.sling.models.annotations.injectorspecific.ValueMapValue;

import javax.annotation.PostConstruct;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.stream.Collectors;

/**
 * Sling Model for Grants Table Component
 * 
 * This model manages a data table displaying grant information with configurable rows.
 * Authors can add/edit grant entries through a multifield dialog.
 * 
 * Features:
 * - Optional section heading
 * - Configurable color scheme (header, rows, text)
 * - Dynamic grant entries (name, type, deadline, amount)
 * - Validation for required fields
 * - Empty state handling
 * 
 * @author AEM Component Engineer
 */
@Model(adaptables = {SlingHttpServletRequest.class, Resource.class},
       adapters = GrantsTableModel.class,
       defaultInjectionStrategy = DefaultInjectionStrategy.OPTIONAL)
public class GrantsTableModel {

    /**
     * Optional heading text for the grants table section
     */
    @ValueMapValue
    private String heading;

    /**
     * Background color for table header
     * Default: #F3F4F6 (light gray)
     */
    @ValueMapValue
    @Default(values = "#F3F4F6")
    private String headerBg;

    /**
     * Background color for table rows
     * Default: #FFFFFF (white)
     */
    @ValueMapValue
    @Default(values = "#FFFFFF")
    private String rowBg;

    /**
     * Primary text color
     * Default: #000000 (black)
     */
    @ValueMapValue
    @Default(values = "#000000")
    private String textColor;

    /**
     * Secondary text color (used for type column)
     * Default: #6B7280 (gray)
     */
    @ValueMapValue
    @Default(values = "#6B7280")
    private String secondaryColor;

    /**
     * List of grant entries configured by authors
     * Each resource contains: grantName, type, deadline, amount
     */
    @ChildResource
    private List<Resource> grants;

    /**
     * Validated and processed list of grant items
     */
    private List<GrantItem> validatedGrants;

    /**
     * Post-construct initialization
     * Validates and processes grant entries, filtering out invalid items
     */
    @PostConstruct
    protected void init() {
        if (grants != null && !grants.isEmpty()) {
            validatedGrants = grants.stream()
                    .map(GrantItem::new)
                    .filter(GrantItem::isValid)
                    .collect(Collectors.toList());
        } else {
            validatedGrants = Collections.emptyList();
        }
    }

    /**
     * Gets the optional section heading
     * 
     * @return the heading text, or null if not configured
     */
    public String getHeading() {
        return heading;
    }

    /**
     * Gets the header background color
     * 
     * @return hex color code for table header background
     */
    public String getHeaderBg() {
        return headerBg;
    }

    /**
     * Gets the row background color
     * 
     * @return hex color code for table row background
     */
    public String getRowBg() {
        return rowBg;
    }

    /**
     * Gets the primary text color
     * 
     * @return hex color code for primary text
     */
    public String getTextColor() {
        return textColor;
    }

    /**
     * Gets the secondary text color
     * 
     * @return hex color code for secondary text (type column)
     */
    public String getSecondaryColor() {
        return secondaryColor;
    }

    /**
     * Gets the list of validated grant entries
     * 
     * @return list of GrantItem objects, empty list if none configured or all invalid
     */
    public List<GrantItem> getGrants() {
        return validatedGrants != null ? validatedGrants : Collections.emptyList();
    }

    /**
     * Gets the count of valid grant entries
     * 
     * @return number of grants in the table
     */
    public int getGrantsCount() {
        return validatedGrants != null ? validatedGrants.size() : 0;
    }

    /**
     * Checks if the table has any grant entries
     * 
     * @return true if at least one grant exists, false otherwise
     */
    public boolean hasGrants() {
        return getGrantsCount() > 0;
    }

    /**
     * Inner class representing a single grant entry
     * Contains all information for one table row
     */
    public class GrantItem {
        private final Resource resource;
        private final String grantName;
        private final String type;
        private final String deadline;
        private final String amount;

        /**
         * Constructor that extracts properties from the resource
         * 
         * @param resource the child resource containing grant data
         */
        public GrantItem(Resource resource) {
            this.resource = resource;
            this.grantName = resource.getValueMap().get("grantName", String.class);
            this.type = resource.getValueMap().get("type", String.class);
            this.deadline = resource.getValueMap().get("deadline", String.class);
            this.amount = resource.getValueMap().get("amount", String.class);
        }

        /**
         * Validates that all required fields are present
         * A grant is considered valid if it has at least a grant name
         * 
         * @return true if grant has required data, false otherwise
         */
        public boolean isValid() {
            return grantName != null && !grantName.trim().isEmpty();
        }

        /**
         * Gets the grant name
         * 
         * @return the name/title of the grant
         */
        public String getGrantName() {
            return grantName;
        }

        /**
         * Gets the grant type
         * Examples: "Competitive", "Partnership", "Direct Award"
         * 
         * @return the type/category of the grant
         */
        public String getType() {
            return type;
        }

        /**
         * Gets the application deadline
         * Format: "MMM DD, YYYY" (e.g., "Dec 31, 2025")
         * 
         * @return the deadline date as a formatted string
         */
        public String getDeadline() {
            return deadline;
        }

        /**
         * Gets the grant amount
         * Format: Currency with appropriate formatting (e.g., "$500,000")
         * 
         * @return the grant amount as a formatted string
         */
        public String getAmount() {
            return amount;
        }

        /**
         * Gets the underlying resource
         * 
         * @return the Sling resource for this grant item
         */
        public Resource getResource() {
            return resource;
        }
    }
}
