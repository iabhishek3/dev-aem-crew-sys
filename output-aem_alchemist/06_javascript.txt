/**
 * Two Panel Info Section Component
 * Handles lazy loading, scroll animations, and accessibility
 */
(function() {
  'use strict';

  // ==================== CONFIGURATION ====================
  var COMPONENT_SELECTOR = '.cmp-two-panel-info';
  var IMAGE_SELECTOR = '.cmp-two-panel-info__image';
  var PANEL_SELECTOR = '.cmp-two-panel-info__panel';
  var IMAGE_CONTAINER_SELECTOR = '.cmp-two-panel-info__image-container';

  // Animation classes
  var ANIMATE_CLASS = 'cmp-two-panel-info--animate';
  var VISIBLE_CLASS = 'cmp-two-panel-info--visible';
  var PANEL_VISIBLE_CLASS = 'cmp-two-panel-info__panel--visible';

  // Configuration
  var config = {
    lazyLoadThreshold: 0.1,
    animationThreshold: 0.15,
    animationDelay: 200,
    imageLoadTimeout: 10000
  };

  // ==================== UTILITY FUNCTIONS ====================

  /**
   * Debounce function for performance optimization
   * @param {Function} func - Function to debounce
   * @param {number} wait - Wait time in milliseconds
   * @return {Function} Debounced function
   */
  function debounce(func, wait) {
    var timeout;
    return function() {
      var args = arguments;
      clearTimeout(timeout);
      timeout = setTimeout(function() {
        func.apply(this, args);
      }.bind(this), wait);
    };
  }

  /**
   * Check if IntersectionObserver is supported
   * @return {boolean} True if supported
   */
  function isIntersectionObserverSupported() {
    return 'IntersectionObserver' in window &&
           'IntersectionObserverEntry' in window &&
           'intersectionRatio' in window.IntersectionObserverEntry.prototype;
  }

  /**
   * Add class to element safely
   * @param {Element} element - DOM element
   * @param {string} className - Class name to add
   */
  function addClass(element, className) {
    if (element && element.classList) {
      element.classList.add(className);
    }
  }

  /**
   * Remove class from element safely
   * @param {Element} element - DOM element
   * @param {string} className - Class name to remove
   */
  function removeClass(element, className) {
    if (element && element.classList) {
      element.classList.remove(className);
    }
  }

  /**
   * Check if element has class
   * @param {Element} element - DOM element
   * @param {string} className - Class name to check
   * @return {boolean} True if element has class
   */
  function hasClass(element, className) {
    return element && element.classList && element.classList.contains(className);
  }

  // ==================== LAZY LOADING ====================

  /**
   * Initialize lazy loading for images
   * @param {Element} component - Component root element
   */
  function initializeLazyLoading(component) {
    if (!isIntersectionObserverSupported()) {
      loadAllImages(component);
      return;
    }

    var images = component.querySelectorAll(IMAGE_SELECTOR);
    if (!images || images.length === 0) {
      return;
    }

    var imageObserver = new IntersectionObserver(
      function(entries) {
        for (var i = 0; i < entries.length; i++) {
          var entry = entries[i];
          if (entry.isIntersecting) {
            loadImage(entry.target);
            imageObserver.unobserve(entry.target);
          }
        }
      },
      {
        rootMargin: '50px',
        threshold: config.lazyLoadThreshold
      }
    );

    for (var i = 0; i < images.length; i++) {
      var img = images[i];
      if (img.dataset.src) {
        imageObserver.observe(img);
      } else {
        imageObserver.observe(img);
      }
    }
  }

  /**
   * Load single image
   * @param {Element} img - Image element
   */
  function loadImage(img) {
    if (!img) {
      return;
    }

    var dataSrc = img.dataset.src;
    if (dataSrc) {
      img.src = dataSrc;
      img.removeAttribute('data-src');
    }

    var container = img.closest(IMAGE_CONTAINER_SELECTOR);
    if (container) {
      addClass(container, 'cmp-two-panel-info__image-container--loading');
    }

    img.addEventListener('load', function() {
      if (container) {
        removeClass(container, 'cmp-two-panel-info__image-container--loading');
        addClass(container, 'cmp-two-panel-info__image-container--loaded');
      }
      img.removeAttribute('data-src');
    });

    img.addEventListener('error', function() {
      console.warn('Failed to load image:', img.src || dataSrc);
      if (container) {
        removeClass(container, 'cmp-two-panel-info__image-container--loading');
        addClass(container, 'cmp-two-panel-info__image-container--error');
      }
      handleImageError(img);
    });

    setTimeout(function() {
      if (container && !hasClass(container, 'cmp-two-panel-info__image-container--loaded')) {
        console.warn('Image load timeout:', img.src || dataSrc);
        handleImageError(img);
      }
    }, config.imageLoadTimeout);
  }

  /**
   * Load all images immediately (fallback for older browsers)
   * @param {Element} component - Component root element
   */
  function loadAllImages(component) {
    var images = component.querySelectorAll(IMAGE_SELECTOR);
    for (var i = 0; i < images.length; i++) {
      loadImage(images[i]);
    }
  }

  /**
   * Handle image loading error
   * @param {Element} img - Image element
   */
  function handleImageError(img) {
    var container = img.closest(IMAGE_CONTAINER_SELECTOR);
    if (container && img.alt) {
      var fallback = document.createElement('div');
      fallback.className = 'cmp-two-panel-info__image-fallback';
      fallback.textContent = img.alt;
      fallback.setAttribute('role', 'img');
      fallback.setAttribute('aria-label', img.alt);
      container.appendChild(fallback);
      img.style.display = 'none';
    }
  }

  // ==================== SCROLL ANIMATIONS ====================

  /**
   * Initialize scroll-based animations
   * @param {Element} component - Component root element
   */
  function initializeScrollAnimations(component) {
    if (!isIntersectionObserverSupported()) {
      addClass(component, VISIBLE_CLASS);
      var panels = component.querySelectorAll(PANEL_SELECTOR);
      for (var i = 0; i < panels.length; i++) {
        addClass(panels[i], PANEL_VISIBLE_CLASS);
      }
      return;
    }

    var panels = component.querySelectorAll(PANEL_SELECTOR);
    if (!panels || panels.length === 0) {
      return;
    }

    var animationObserver = new IntersectionObserver(
      function(entries) {
        for (var i = 0; i < entries.length; i++) {
          var entry = entries[i];
          if (entry.isIntersecting) {
            animatePanel(entry.target);
            animationObserver.unobserve(entry.target);
          }
        }
      },
      {
        rootMargin: '0px',
        threshold: config.animationThreshold
      }
    );

    addClass(component, ANIMATE_CLASS);

    for (var i = 0; i < panels.length; i++) {
      animationObserver.observe(panels[i]);
    }
  }

  /**
   * Animate panel into view
   * @param {Element} panel - Panel element
   */
  function animatePanel(panel) {
    if (!panel) {
      return;
    }

    setTimeout(function() {
      addClass(panel, PANEL_VISIBLE_CLASS);
    }, config.animationDelay);
  }

  // ==================== ACCESSIBILITY ====================

  /**
   * Initialize accessibility features
   * @param {Element} component - Component root element
   */
  function initializeAccessibility(component) {
    var imageContainers = component.querySelectorAll(IMAGE_CONTAINER_SELECTOR);
    
    for (var i = 0; i < imageContainers.length; i++) {
      var container = imageContainers[i];
      var img = container.querySelector(IMAGE_SELECTOR);
      
      if (img && img.alt) {
        container.setAttribute('tabindex', '0');
        container.setAttribute('role', 'img');
        container.setAttribute('aria-label', img.alt);
        container.addEventListener('keydown', handleImageKeydown);
      }
    }

    component.setAttribute('role', 'region');
    component.setAttribute('aria-label', 'Informational section with image panels');
  }

  /**
   * Handle keyboard navigation for images
   * @param {KeyboardEvent} event - Keyboard event
   */
  function handleImageKeydown(event) {
    if (event.key === 'Enter' || event.key === ' ') {
      event.preventDefault();
      var container = event.currentTarget;
      var img = container.querySelector(IMAGE_SELECTOR);
      
      if (img) {
        img.focus();
      }
    }
  }

  // ==================== RESIZE HANDLER ====================

  /**
   * Handle window resize
   * @param {Element} component - Component root element
   */
  function handleResize(component) {
    // Placeholder for future dynamic positioning logic
  }

  /**
   * Create debounced resize handler
   * @param {Element} component - Component root element
   * @return {Function} Debounced resize handler
   */
  function createResizeHandler(component) {
    return debounce(function() {
      handleResize(component);
    }, 250);
  }

  // ==================== COMPONENT INITIALIZATION ====================

  /**
   * Initialize single component instance
   * @param {Element} component - Component root element
   */
  function initializeComponent(component) {
    if (!component) {
      console.warn('Two Panel Info Section: Invalid component element');
      return;
    }

    if (component.dataset.initialized === 'true') {
      return;
    }

    try {
      initializeLazyLoading(component);
      initializeScrollAnimations(component);
      initializeAccessibility(component);

      var resizeHandler = createResizeHandler(component);
      window.addEventListener('resize', resizeHandler);

      component.dataset.initialized = 'true';
      component.dataset.resizeHandler = 'attached';

      if (window.console && window.console.log) {
        console.log('Two Panel Info Section initialized successfully');
      }

    } catch (error) {
      console.error('Error initializing Two Panel Info Section:', error);
    }
  }

  /**
   * Initialize all component instances on the page
   */
  function init() {
    var components = document.querySelectorAll(COMPONENT_SELECTOR);
    
    if (!components || components.length === 0) {
      return;
    }

    for (var i = 0; i < components.length; i++) {
      initializeComponent(components[i]);
    }
  }

  // ==================== AUTO-INITIALIZATION ====================

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }

  if (window.MutationObserver) {
    var observer = new MutationObserver(function(mutations) {
      for (var i = 0; i < mutations.length; i++) {
        var mutation = mutations[i];
        var addedNodes = mutation.addedNodes;
        
        for (var j = 0; j < addedNodes.length; j++) {
          var node = addedNodes[j];
          
          if (node.nodeType === 1) {
            if (node.matches && node.matches(COMPONENT_SELECTOR)) {
              initializeComponent(node);
            }
            
            var nestedComponents = node.querySelectorAll(COMPONENT_SELECTOR);
            for (var k = 0; k < nestedComponents.length; k++) {
              initializeComponent(nestedComponents[k]);
            }
          }
        }
      }
    });

    observer.observe(document.body, {
      childList: true,
      subtree: true
    });
  }

})();
