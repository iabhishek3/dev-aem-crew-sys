/**
 * Grants Table Component JavaScript
 * 
 * Component Type: Data Table
 * Features: Keyboard navigation, accessibility, analytics tracking
 * ES5 Compatible: Uses var, function syntax (no const/let/arrow functions)
 */

(function() {
  'use strict';

  // ============================================================================
  // Constants and Configuration
  // ============================================================================

  var COMPONENT_SELECTOR = '.cmp-grants-table';
  var TABLE_SELECTOR = '.cmp-grants-table__table';
  var TBODY_SELECTOR = '.cmp-grants-table__tbody';
  var ROW_SELECTOR = '.cmp-grants-table__row';
  var WRAPPER_SELECTOR = '.cmp-grants-table__wrapper';
  
  // Keyboard codes
  var KEY_CODES = {
    ENTER: 13,
    SPACE: 32,
    ARROW_UP: 38,
    ARROW_DOWN: 40,
    HOME: 36,
    END: 35
  };

  // ============================================================================
  // Initialization
  // ============================================================================

  /**
   * Initialize all grants table components on the page
   */
  function init() {
    var components = document.querySelectorAll(COMPONENT_SELECTOR);
    
    if (components.length === 0) {
      return;
    }
    
    // Initialize each component instance
    for (var i = 0; i < components.length; i++) {
      initializeComponent(components[i]);
    }
  }

  /**
   * Initialize a single grants table component
   * @param {HTMLElement} component - The grants table component element
   */
  function initializeComponent(component) {
    try {
      var table = component.querySelector(TABLE_SELECTOR);
      var wrapper = component.querySelector(WRAPPER_SELECTOR);
      var tbody = component.querySelector(TBODY_SELECTOR);
      
      if (!table || !tbody) {
        console.warn('Grants Table: Required elements not found in component');
        return;
      }
      
      // Initialize features
      initializeAccessibility(component, table, tbody);
      initializeScrollBehavior(wrapper);
      initializeRowInteractions(tbody);
      
      // Mark component as initialized
      component.setAttribute('data-cmp-initialized', 'true');
      
    } catch (error) {
      console.error('Grants Table: Error initializing component:', error);
    }
  }

  // ============================================================================
  // Accessibility Features
  // ============================================================================

  /**
   * Initialize accessibility features for keyboard navigation
   * @param {HTMLElement} component - The component container
   * @param {HTMLElement} table - The table element
   * @param {HTMLElement} tbody - The table body element
   */
  function initializeAccessibility(component, table, tbody) {
    // Make table focusable for keyboard navigation
    table.setAttribute('tabindex', '0');
    table.setAttribute('role', 'table');
    table.setAttribute('aria-label', 'Grants table');
    
    // Add keyboard navigation
    table.addEventListener('keydown', function(event) {
      handleKeyboardNavigation(event, tbody);
    });
    
    // Update ARIA attributes for rows
    var rows = tbody.querySelectorAll(ROW_SELECTOR);
    for (var i = 0; i < rows.length; i++) {
      var row = rows[i];
      if (!row.classList.contains('cmp-grants-table__row--empty')) {
        row.setAttribute('role', 'row');
        row.setAttribute('tabindex', '-1');
      }
    }
  }

  /**
   * Handle keyboard navigation within the table
   * @param {KeyboardEvent} event - The keyboard event
   * @param {HTMLElement} tbody - The table body element
   */
  function handleKeyboardNavigation(event, tbody) {
    var keyCode = event.keyCode || event.which;
    var rows = tbody.querySelectorAll(ROW_SELECTOR + ':not(.cmp-grants-table__row--empty)');
    
    if (rows.length === 0) {
      return;
    }
    
    var currentRow = document.activeElement;
    var currentIndex = -1;
    
    // Find current row index
    for (var i = 0; i < rows.length; i++) {
      if (rows[i] === currentRow || rows[i].contains(currentRow)) {
        currentIndex = i;
        break;
      }
    }
    
    var handled = false;
    var targetRow = null;
    
    switch (keyCode) {
      case KEY_CODES.ARROW_DOWN:
        // Move to next row
        if (currentIndex < rows.length - 1) {
          targetRow = rows[currentIndex + 1];
          handled = true;
        }
        break;
        
      case KEY_CODES.ARROW_UP:
        // Move to previous row
        if (currentIndex > 0) {
          targetRow = rows[currentIndex - 1];
          handled = true;
        }
        break;
        
      case KEY_CODES.HOME:
        // Move to first row
        targetRow = rows[0];
        handled = true;
        break;
        
      case KEY_CODES.END:
        // Move to last row
        targetRow = rows[rows.length - 1];
        handled = true;
        break;
    }
    
    if (handled && targetRow) {
      event.preventDefault();
      focusRow(targetRow);
    }
  }

  /**
   * Focus a table row and update ARIA attributes
   * @param {HTMLElement} row - The row to focus
   */
  function focusRow(row) {
    if (!row) {
      return;
    }
    
    row.focus();
    
    // Announce row to screen readers
    var cells = row.querySelectorAll('.cmp-grants-table__cell');
    var rowText = '';
    for (var i = 0; i < cells.length; i++) {
      rowText += cells[i].textContent.trim() + ', ';
    }
    
    announceToScreenReader('Selected row: ' + rowText);
  }

  /**
   * Announce text to screen readers using ARIA live region
   * @param {string} message - The message to announce
   */
  function announceToScreenReader(message) {
    var liveRegion = document.getElementById('grants-table-live-region');
    
    if (!liveRegion) {
      liveRegion = document.createElement('div');
      liveRegion.id = 'grants-table-live-region';
      liveRegion.setAttribute('aria-live', 'polite');
      liveRegion.setAttribute('aria-atomic', 'true');
      liveRegion.style.position = 'absolute';
      liveRegion.style.left = '-10000px';
      liveRegion.style.width = '1px';
      liveRegion.style.height = '1px';
      liveRegion.style.overflow = 'hidden';
      document.body.appendChild(liveRegion);
    }
    
    liveRegion.textContent = message;
  }

  // ============================================================================
  // Scroll Behavior
  // ============================================================================

  /**
   * Initialize scroll behavior for responsive tables
   * @param {HTMLElement} wrapper - The table wrapper element
   */
  function initializeScrollBehavior(wrapper) {
    if (!wrapper) {
      return;
    }
    
    // Add shadow indicators when content is scrollable
    updateScrollIndicators(wrapper);
    
    wrapper.addEventListener('scroll', function() {
      updateScrollIndicators(wrapper);
    });
    
    // Update indicators on window resize
    window.addEventListener('resize', debounce(function() {
      updateScrollIndicators(wrapper);
    }, 150));
  }

  /**
   * Update visual indicators for scrollable content
   * @param {HTMLElement} wrapper - The table wrapper element
   */
  function updateScrollIndicators(wrapper) {
    var scrollLeft = wrapper.scrollLeft;
    var scrollWidth = wrapper.scrollWidth;
    var clientWidth = wrapper.clientWidth;
    var maxScroll = scrollWidth - clientWidth;
    
    // Add class if scrolled from left
    if (scrollLeft > 0) {
      wrapper.classList.add('cmp-grants-table__wrapper--scrolled-left');
    } else {
      wrapper.classList.remove('cmp-grants-table__wrapper--scrolled-left');
    }
    
    // Add class if not scrolled to right end
    if (scrollLeft < maxScroll - 1) {
      wrapper.classList.add('cmp-grants-table__wrapper--scrolled-right');
    } else {
      wrapper.classList.remove('cmp-grants-table__wrapper--scrolled-right');
    }
  }

  // ============================================================================
  // Row Interactions
  // ============================================================================

  /**
   * Initialize row interaction tracking
   * @param {HTMLElement} tbody - The table body element
   */
  function initializeRowInteractions(tbody) {
    var rows = tbody.querySelectorAll(ROW_SELECTOR + ':not(.cmp-grants-table__row--empty)');
    
    for (var i = 0; i < rows.length; i++) {
      var row = rows[i];
      
      // Track row clicks for analytics
      row.addEventListener('click', handleRowClick);
      
      // Track keyboard interactions
      row.addEventListener('keydown', function(event) {
        var keyCode = event.keyCode || event.which;
        if (keyCode === KEY_CODES.ENTER || keyCode === KEY_CODES.SPACE) {
          event.preventDefault();
          handleRowClick.call(this, event);
        }
      });
    }
  }

  /**
   * Handle row click events
   * @param {Event} event - The click event
   */
  function handleRowClick(event) {
    var row = event.currentTarget;
    var cells = row.querySelectorAll('.cmp-grants-table__cell');
    
    if (cells.length === 0) {
      return;
    }
    
    // Extract row data
    var grantName = cells[0] ? cells[0].textContent.trim() : '';
    var grantType = cells[1] ? cells[1].textContent.trim() : '';
    var deadline = cells[2] ? cells[2].textContent.trim() : '';
    var amount = cells[3] ? cells[3].textContent.trim() : '';
    
    // Log interaction for analytics
    logAnalytics('grant_row_click', {
      grantName: grantName,
      type: grantType,
      deadline: deadline,
      amount: amount
    });
  }

  /**
   * Log analytics events
   * @param {string} eventName - The event name
   * @param {Object} data - The event data
   */
  function logAnalytics(eventName, data) {
    // Check for Adobe Analytics (DTM/Launch)
    if (typeof window.digitalData !== 'undefined') {
      try {
        window.digitalData.events = window.digitalData.events || [];
        window.digitalData.events.push({
          eventName: eventName,
          eventData: data
        });
      } catch (error) {
        console.warn('Grants Table: Analytics tracking error:', error);
      }
    }
    
    // Fallback: Log to console in development
    console.log('Grants Table Analytics:', eventName, data);
  }

  // ============================================================================
  // Utility Functions
  // ============================================================================

  /**
   * Debounce function to limit event handler execution
   * @param {Function} func - The function to debounce
   * @param {number} wait - The wait time in milliseconds
   * @return {Function} The debounced function
   */
  function debounce(func, wait) {
    var timeout;
    return function() {
      var context = this;
      var args = arguments;
      
      clearTimeout(timeout);
      timeout = setTimeout(function() {
        func.apply(context, args);
      }, wait);
    };
  }

  // ============================================================================
  // Auto-initialization
  // ============================================================================

  // Initialize when DOM is ready
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    // DOM already loaded
    init();
  }

  // Re-initialize on AEM author refresh (for author mode)
  if (typeof Granite !== 'undefined' && Granite.author) {
    Granite.author.ui.onRefresh(function() {
      init();
    });
  }

})();
